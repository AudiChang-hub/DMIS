```text
CASE
 WHEN REGEXP_MATCH(Model, r"^EV0(62|60L|76|70V|76S|76SZV)|Gogoro|Pulse|S2 ABS|S2ABS") THEN "白牌電車"
 WHEN REGEXP_MATCH(Model, r"JEGO|VIVA|EZ1|EZZY|Ur2") THEN "綠牌電車"
 WHEN REGEXP_MATCH(Model, r"BOBE|SHINE|TSV57|M02|M01|JDL-B1") THEN "微型電車"
 WHEN REGEXP_MATCH(Model, r"UQ|UC|UG|UT|UT125XZ") THEN "速克達"
 WHEN REGEXP_MATCH(Model, r"DRZ-4SM|GSX|DS|DS250") THEN "擋車"

END
```
